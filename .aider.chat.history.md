import numpy as np
from pydantic import BaseModel
from typing import Literal

class MemoryNode(BaseModel):
    id: str
    type: Literal["event", "thought", "chat"]
    subject: str
    predicate: str
    object: str
    embedding: list[float]
    poignancy: float
    created: datetime
    accessed: datetime

class VectorMemory:
    def __init__(self, db_path: str = ":memory:"):
        self.db = sqlite3.connect(db_path)
        self._init_schema()
        
    def _init_schema(self):
        self.db.execute("""
            CREATE TABLE IF NOT EXISTS memories (
                id TEXT PRIMARY KEY,
                type TEXT NOT NULL,
                subject TEXT,
                predicate TEXT,
                object TEXT,
                embedding BLOB,
                poignancy REAL,
                created TIMESTAMP,
                accessed TIMESTAMP
            )
        """)
        
    def add_memory(self, node: MemoryNode):
        # Store embedding as bytes for efficient storage
        embedding_bytes = np.asarray(node.embedding).tobytes()
        self.db.execute(
            """
            INSERT INTO memories VALUES (
                ?, ?, ?, ?, ?, ?, ?, ?, ?
            )
            """,
            (
                node.id,
                node.type,
                node.subject,
                node.predicate,
                node.object,
                embedding_bytes,
                node.poignancy,
                node.created.isoformat(),
                node.accessed.isoformat()
            )
